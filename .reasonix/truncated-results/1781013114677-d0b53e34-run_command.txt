552:552:/// (compatible with Claude Code's path + block format).
553:553:fn parse_search_replace_blocks(input: &str) -> Result<Vec<(String, String)>, String> {
554:554:    let mut blocks = Vec::new();
555:555:    let mut remaining = input;
556:556:
557:557:    loop {
558:558:        // Find next "<<<<<<< SEARCH" marker
559:559:        let search_start = match remaining.find("<<<<<<< SEARCH") {
560:560:            Some(idx) => idx,
561:561:            None => break,
562:562:        };
563:563:
564:564:        // Content after the marker
565:565:        let after_marker = &remaining[search_start + 14..];
566:566:        // Strip leading newline (the marker is usually followed by \n)
567:567:        let body = after_marker.strip_prefix('\n').unwrap_or(after_marker);
568:568:
569:569:        // Find the "=======" separator
570:570:        // Normal case: "\n=======\n" within body
571:571:        // Empty SEARCH case: body starts with "=======\n"
572:572:        let (search_text, rest) = if let Some(idx) = body.find("\n=======\n") {
573:573:            (&body[..idx], &body[idx + 10..])
574:574:        } else if body.starts_with("=======\n") {
575:575:            // Empty SEARCH (create file / full replace)
576:576:            ("", &body[9..])
577:577:        } else {

580:580:            ));
581:581:        };
582:582:
583:583:        // Find the ">>>>>>> REPLACE" end marker
584:584:        let end_idx = match rest.find("\n>>>>>>> REPLACE") {
585:585:            Some(idx) => idx,
586:586:            None => {
587:587:                return Err(format!("SEARCH/REPLACE 块缺少 '>>>>>>> REPLACE' 结束标记"));
588:588:            }
589:589:        };
590:590:
591:591:        let replace_text = &rest[..end_idx];
592:592:        blocks.push((search_text.to_string(), replace_text.to_string()));
593:593:

595:595:        remaining = &rest[end_idx + 15..]; // skip "\n>>>>>>> REPLACE"
